cmake_minimum_required(VERSION 3.20)
project(recomp_mod_core VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
add_library(recomp_mod_core STATIC src/runtime.cpp src/rmod_crc32.c src/rmod_sha256.c)
add_library(recomp-mod-core::core ALIAS recomp_mod_core)
set_target_properties(recomp_mod_core PROPERTIES EXPORT_NAME core)
target_compile_features(recomp_mod_core PUBLIC cxx_std_17)
target_include_directories(recomp_mod_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src)
if(MSVC)
  target_compile_options(recomp_mod_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(recomp_mod_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(rmod_test tests/runtime_test.cpp)
  target_link_libraries(rmod_test PRIVATE recomp_mod_core)
  add_test(NAME package_and_transactions COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures.py $<TARGET_FILE:rmod_test> ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  add_executable(rmod_c_api_test tests/c_api.c)
  target_link_libraries(rmod_c_api_test PRIVATE recomp_mod_core)
  add_test(NAME c_abi COMMAND rmod_c_api_test)
  add_executable(rmod_qualified_api_test tests/qualified_api_test.cpp)
  target_link_libraries(rmod_qualified_api_test PRIVATE recomp_mod_core)
  if(MSVC)
    target_compile_options(rmod_qualified_api_test PRIVATE /utf-8)
  endif()
  add_test(NAME qualified_apis_and_archives COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/qualified_fixtures.py $<TARGET_FILE:rmod_qualified_api_test> ${CMAKE_CURRENT_BINARY_DIR}/qualified-fixtures)
endif()
install(TARGETS recomp_mod_core EXPORT recomp-mod-core-targets ARCHIVE DESTINATION lib)
install(FILES include/recomp_mod.h DESTINATION include)
install(FILES include/recomp_mod/archive.hpp DESTINATION include/recomp_mod)
install(EXPORT recomp-mod-core-targets NAMESPACE recomp-mod-core:: DESTINATION lib/cmake/recomp-mod-core)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/recomp-mod-core-config.cmake "include(\"\${CMAKE_CURRENT_LIST_DIR}/recomp-mod-core-targets.cmake\")\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/recomp-mod-core-config.cmake DESTINATION lib/cmake/recomp-mod-core)
